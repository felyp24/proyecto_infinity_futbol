CREATE DATABASE IF NOT EXISTS infinity_futbol_db
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE infinity_futbol_db;

-- =========================================================
-- TABLA: distrito
-- =========================================================
CREATE TABLE distrito (
    id_distrito VARCHAR(10) PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    estado TINYINT NOT NULL DEFAULT 1
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: sede
-- =========================================================
CREATE TABLE sede (
    id_sede VARCHAR(20) PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    direccion VARCHAR(150) NOT NULL,
    referencia VARCHAR(150),
    id_distrito VARCHAR(10) NOT NULL,
    estado TINYINT NOT NULL DEFAULT 1,

    CONSTRAINT fk_sede_distrito
        FOREIGN KEY (id_distrito) REFERENCES distrito(id_distrito)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: cancha
-- =========================================================
CREATE TABLE cancha (
    id_cancha VARCHAR(20) PRIMARY KEY,
    numero_cancha INT NOT NULL,
    tipo_superficie VARCHAR(50),
    estado VARCHAR(30) NOT NULL DEFAULT 'DISPONIBLE',
    id_sede VARCHAR(20) NOT NULL,

    CONSTRAINT fk_cancha_sede
        FOREIGN KEY (id_sede) REFERENCES sede(id_sede)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: rol
-- =========================================================
CREATE TABLE rol (
    id_rol VARCHAR(20) PRIMARY KEY,
    nombre_rol VARCHAR(50) NOT NULL UNIQUE,
    descripcion VARCHAR(150),
    estado TINYINT NOT NULL DEFAULT 1
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: usuario
-- =========================================================
CREATE TABLE usuario (
    id_usuario VARCHAR(20) PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    correo VARCHAR(100) NOT NULL UNIQUE,
    estado VARCHAR(30) NOT NULL DEFAULT 'ACTIVO',
    fecha_creacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ultimo_acceso DATETIME NULL
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: usuario_rol
-- =========================================================
CREATE TABLE usuario_rol (
    id_usuario_rol VARCHAR(25) PRIMARY KEY,
    id_usuario VARCHAR(20) NOT NULL,
    id_rol VARCHAR(20) NOT NULL,

    CONSTRAINT fk_usuario_rol_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario),

    CONSTRAINT fk_usuario_rol_rol
        FOREIGN KEY (id_rol) REFERENCES rol(id_rol),

    CONSTRAINT uk_usuario_rol UNIQUE (id_usuario, id_rol)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: alumno
-- =========================================================
CREATE TABLE alumno (
    id_alumno VARCHAR(20) PRIMARY KEY,
    id_usuario VARCHAR(20) NOT NULL UNIQUE,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    tipo_documento VARCHAR(30) NOT NULL,
    numero_documento VARCHAR(20) NOT NULL UNIQUE,
    fecha_nacimiento DATE,
    telefono VARCHAR(20),
    fecha_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    estado VARCHAR(30) NOT NULL DEFAULT 'ACTIVO',

    CONSTRAINT fk_alumno_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: entrenador
-- =========================================================
CREATE TABLE entrenador (
    id_entrenador VARCHAR(20) PRIMARY KEY,
    id_usuario VARCHAR(20) NOT NULL UNIQUE,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    telefono VARCHAR(20),
    especialidad VARCHAR(100),
    estado VARCHAR(30) NOT NULL DEFAULT 'ACTIVO',

    CONSTRAINT fk_entrenador_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: clase
-- =========================================================
CREATE TABLE clase (
    id_clase VARCHAR(20) PRIMARY KEY,
    titulo VARCHAR(100) NOT NULL,
    descripcion VARCHAR(255),
    fecha_clase DATE NOT NULL,
    hora_inicio TIME NOT NULL,
    hora_fin TIME NOT NULL,
    cupo_maximo INT NOT NULL,
    cupo_disponible INT NOT NULL,
    estado VARCHAR(30) NOT NULL DEFAULT 'PROGRAMADA',
    id_cancha VARCHAR(20) NOT NULL,
    id_entrenador VARCHAR(20) NOT NULL,

    CONSTRAINT fk_clase_cancha
        FOREIGN KEY (id_cancha) REFERENCES cancha(id_cancha),

    CONSTRAINT fk_clase_entrenador
        FOREIGN KEY (id_entrenador) REFERENCES entrenador(id_entrenador)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: cuenta_credito
-- =========================================================
CREATE TABLE cuenta_credito (
    id_cuenta_credito VARCHAR(20) PRIMARY KEY,
    id_alumno VARCHAR(20) NOT NULL UNIQUE,
    saldo_actual INT NOT NULL DEFAULT 0,
    fecha_actualizacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_cuenta_credito_alumno
        FOREIGN KEY (id_alumno) REFERENCES alumno(id_alumno)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: paquete_credito
-- =========================================================
CREATE TABLE paquete_credito (
    id_paquete_credito VARCHAR(20) PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    cantidad_creditos INT NOT NULL,
    precio DECIMAL(10,2) NOT NULL,
    dias_vigencia INT NOT NULL,
    estado TINYINT NOT NULL DEFAULT 1
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: cupon_descuento
-- =========================================================
CREATE TABLE cupon_descuento (
    id_cupon VARCHAR(20) PRIMARY KEY,
    codigo VARCHAR(50) NOT NULL UNIQUE,
    porcentaje_descuento DECIMAL(5,2) NOT NULL,
    fecha_inicio DATE NOT NULL,
    fecha_expiracion DATE NOT NULL,
    estado VARCHAR(30) NOT NULL DEFAULT 'ACTIVO'
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: pago
-- =========================================================
CREATE TABLE pago (
    id_pago VARCHAR(20) PRIMARY KEY,
    id_alumno VARCHAR(20) NOT NULL,
    id_paquete_credito VARCHAR(20) NOT NULL,
    id_cupon VARCHAR(20) NULL,
    fecha_pago DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    monto_bruto DECIMAL(10,2) NOT NULL,
    monto_descuento DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    monto_total DECIMAL(10,2) NOT NULL,
    metodo_pago VARCHAR(50) NOT NULL,
    estado_pago VARCHAR(30) NOT NULL DEFAULT 'PENDIENTE',

    CONSTRAINT fk_pago_alumno
        FOREIGN KEY (id_alumno) REFERENCES alumno(id_alumno),

    CONSTRAINT fk_pago_paquete_credito
        FOREIGN KEY (id_paquete_credito) REFERENCES paquete_credito(id_paquete_credito),

    CONSTRAINT fk_pago_cupon
        FOREIGN KEY (id_cupon) REFERENCES cupon_descuento(id_cupon)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: reserva
-- =========================================================
CREATE TABLE reserva (
    id_reserva VARCHAR(20) PRIMARY KEY,
    id_alumno VARCHAR(20) NOT NULL,
    id_clase VARCHAR(20) NOT NULL,
    fecha_reserva DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    creditos_usados INT NOT NULL DEFAULT 1,
    estado VARCHAR(30) NOT NULL DEFAULT 'CONFIRMADA',

    CONSTRAINT fk_reserva_alumno
        FOREIGN KEY (id_alumno) REFERENCES alumno(id_alumno),

    CONSTRAINT fk_reserva_clase
        FOREIGN KEY (id_clase) REFERENCES clase(id_clase),

    CONSTRAINT uk_reserva_alumno_clase UNIQUE (id_alumno, id_clase)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: asistencia
-- =========================================================
CREATE TABLE asistencia (
    id_asistencia VARCHAR(20) PRIMARY KEY,
    id_reserva VARCHAR(20) NOT NULL UNIQUE,
    estado_asistencia VARCHAR(30) NOT NULL,
    hora_marcacion DATETIME,
    observacion VARCHAR(255),

    CONSTRAINT fk_asistencia_reserva
        FOREIGN KEY (id_reserva) REFERENCES reserva(id_reserva)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: movimiento_credito
-- =========================================================
CREATE TABLE movimiento_credito (
    id_movimiento_credito VARCHAR(25) PRIMARY KEY,
    id_cuenta_credito VARCHAR(20) NOT NULL,
    id_pago VARCHAR(20) NULL,
    id_reserva VARCHAR(20) NULL,
    tipo_movimiento VARCHAR(30) NOT NULL,
    cantidad INT NOT NULL,
    fecha_movimiento DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_expiracion DATE NULL,
    descripcion VARCHAR(255),

    CONSTRAINT fk_movimiento_credito_cuenta
        FOREIGN KEY (id_cuenta_credito) REFERENCES cuenta_credito(id_cuenta_credito),

    CONSTRAINT fk_movimiento_credito_pago
        FOREIGN KEY (id_pago) REFERENCES pago(id_pago),

    CONSTRAINT fk_movimiento_credito_reserva
        FOREIGN KEY (id_reserva) REFERENCES reserva(id_reserva)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: comprobante
-- Versión simple para boleta/factura
-- =========================================================
CREATE TABLE comprobante (
    id_comprobante VARCHAR(20) PRIMARY KEY,
    id_pago VARCHAR(20) NOT NULL UNIQUE,
    tipo_comprobante VARCHAR(30) NOT NULL,
    serie VARCHAR(10) NOT NULL,
    numero VARCHAR(20) NOT NULL,
    fecha_emision DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    monto_total DECIMAL(10,2) NOT NULL,
    estado VARCHAR(30) NOT NULL DEFAULT 'EMITIDO',

    CONSTRAINT fk_comprobante_pago
        FOREIGN KEY (id_pago) REFERENCES pago(id_pago),

    CONSTRAINT uk_comprobante_serie_numero UNIQUE (serie, numero)
) ENGINE=InnoDB;

-- =========================================================
-- TABLA: notificacion
-- =========================================================
CREATE TABLE notificacion (
    id_notificacion VARCHAR(20) PRIMARY KEY,
    id_alumno VARCHAR(20) NOT NULL,
    id_reserva VARCHAR(20) NULL,
    titulo VARCHAR(100) NOT NULL,
    mensaje VARCHAR(255) NOT NULL,
    tipo VARCHAR(50) NOT NULL,
    fecha_programada DATETIME,
    fecha_envio DATETIME,
    estado VARCHAR(30) NOT NULL DEFAULT 'PENDIENTE',

    CONSTRAINT fk_notificacion_alumno
        FOREIGN KEY (id_alumno) REFERENCES alumno(id_alumno),

    CONSTRAINT fk_notificacion_reserva
        FOREIGN KEY (id_reserva) REFERENCES reserva(id_reserva)
) ENGINE=InnoDB;

-- =========================================================
-- ÍNDICES RECOMENDADOS
-- =========================================================
CREATE INDEX idx_clase_fecha ON clase(fecha_clase);
CREATE INDEX idx_reserva_estado ON reserva(estado);
CREATE INDEX idx_pago_fecha ON pago(fecha_pago);
CREATE INDEX idx_pago_estado ON pago(estado_pago);
CREATE INDEX idx_movimiento_credito_tipo ON movimiento_credito(tipo_movimiento);
CREATE INDEX idx_comprobante_fecha ON comprobante(fecha_emision);
CREATE INDEX idx_notificacion_estado ON notificacion(estado);
